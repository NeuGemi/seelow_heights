# Add Unplanned Offensives
country_event = {
	id = offensives.1

	fire_only_once = yes

	is_triggered_only = yes

	hidden = yes


    option = {
    ai_chance = {
    base = 100
    	}
		hidden_effect = {
			set_global_flag = ALLIES_unplanned_offensives
			every_state = {
				limit = {
					OR = {
						state = 7 # Holland
						state = 763 # Königsberg
						state = 190 # Kurland
						state = 809
						state = 58 # Mecklenburg
						state = 61 # Schleswig-Holstein
						state = 9
						state = 69
						state = 848
						state = 153
						state = 39
						state = 23
						state = 30
						state = 14
						state = 29
					}
				}
				set_state_flag = ALLIES_unplanned_offensive_flag
				add_dynamic_modifier = {
					modifier = unplanned_offensive
				}
			}
		}
    }
}
